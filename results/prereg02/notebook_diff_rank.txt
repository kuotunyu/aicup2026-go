+再按一次「全部執行」就好：
+

+def d4(x, s):
+    # 8 種對稱：s=0 原方向；s>=4 先在 dim 3 鏡射，再在 dims (2, 3) 旋轉 s%4 次（x 是 [batch, 500, 19, 19]）

+        return x
+    y = torch.flip(x, dims=(3,)) if s >= 4 else x
+    return torch.rot90(y, k=s % 4, dims=(2, 3)).contiguous()
